SELECT category_id, SUM(stock) FROM products GROUP BY category_id;
SELECT u.name FROM users u JOIN orders o ON u.id = o.customer_id WHERE o.id = 5;
SELECT SUM(p.amount) FROM payments p JOIN orders o ON p.order_id = o.id JOIN users u ON o.customer_id = u.id WHERE u.name = 'John Doe';
SELECT AVG(r.rating) FROM reviews r WHERE r.product_id = (     SELECT p.id     FROM products p     WHERE p.name = 'Laptop' );
SELECT COUNT(*) FROM user_login_attempts WHERE success_status = 1 AND attempt_time >= datetime('now', '-30 days');
SELECT st.subject FROM support_tickets st JOIN users u ON st.user_id = u.id JOIN user_profiles up ON u.id = up.id WHERE up.theme = 'dark';
SELECT p.name FROM products p JOIN order_items oi ON p.id = oi.product_id JOIN orders o ON oi.order_id = o.id JOIN devices d ON o.customer_id = d.user_id WHERE d.platform = 'Android';
SELECT SUM(cart_items.quantity) FROM cart_items JOIN cart ON cart_items.cart_id = cart.id JOIN users ON cart.user_id = users.id WHERE users.name = 'Jane Smith';
SELECT * FROM coupons WHERE discount_percent > 20 AND active = 1;
